# 直接粘贴到云效「Docker 部署」→ 部署脚本（$image 为上游镜像变量）
# ACR 账密请只写在云效部署框 / 流水线变量，不要提交仓库

echo "$ACR_PASS" | docker login --username="$ACR_USER" --password-stdin registry.cn-hangzhou.aliyuncs.com

CONTAINER_NAME="cm-opc-workbench-admin-ui"
BACKEND_UPSTREAM="106.15.53.78:9080"

docker stop $CONTAINER_NAME || true
docker rm $CONTAINER_NAME || true

for img in `docker images | grep -i opc-workbench-admin-ui | awk '{print $3}'`
do
   echo $img
   docker rmi $img || true
done

docker pull $image

docker run -d --restart=always \
  --name="$CONTAINER_NAME" \
  -p 9002:9002 \
  -e BACKEND_UPSTREAM="$BACKEND_UPSTREAM" \
  $image
